package player

import (
	"context"
	"encoding/json"
	"os"
	"testing"
)

func TestRootCauseSnapshotFailureCategoryLoss(t *testing.T) {
	var rows []map[string]string
	for _, cause := range []string{"audio_network_timeout", "audio_access_denied", "audio_stream_truncated"} {
		p := testPlayer()
		if p.Load(context.Background(), "aaaaaaaaaaa") != nil {
			t.Fatal("fixture_setup_failed")
		}
		p.Pause()
		p.mu.Lock()
		p.wantPlay = true
		p.session.decoder.eof.Store(true)
		p.session.decoder.failed.Store(true)
		p.session.decoder.bridge = &streamBridge{category: cause}
		p.mu.Unlock()
		state := p.Snapshot()
		rows = append(rows, map[string]string{"underlying": cause, "observed": state.Error})
		// Remove the synthetic bridge before normal lifecycle cleanup.
		p.mu.Lock()
		p.session.decoder.bridge = nil
		p.mu.Unlock()
		p.Close()
	}
	if path := os.Getenv("YTMGO_LIVE_EVIDENCE"); path != "" {
		b, _ := json.MarshalIndent(rows, "", "  ")
		_ = os.WriteFile(path, append(b, '\n'), 0600)
	}
	for _, row := range rows {
		if row["observed"] != "audio_stream_failed" {
			t.Fatal("source_behavior_drifted")
		}
	}
}
