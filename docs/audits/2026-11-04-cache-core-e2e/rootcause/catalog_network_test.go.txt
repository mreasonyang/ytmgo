type rootcauseCatalogMetric struct {
	Status   int    `json:"status"`
	HeaderMS int64  `json:"header_ms"`
	TotalMS  int64  `json:"total_ms"`
	Bytes    int    `json:"bytes"`
	Cause    string `json:"cause"`
}
type rootcauseCatalogBody struct {
	io.ReadCloser
	ctx    context.Context
	began  time.Time
	metric *rootcauseCatalogMetric
}

func rootcauseCatalogError(ctx context.Context, e error) string {
	if e == nil || e == io.EOF {
		return "none"
	}
	if errors.Is(ctx.Err(), context.DeadlineExceeded) {
		return "deadline"
	}
	var ne net.Error
	if errors.As(e, &ne) && ne.Timeout() {
		return "network_timeout"
	}
	if errors.Is(e, io.ErrUnexpectedEOF) {
		return "unexpected_eof"
	}
	return "transport_other"
}
func (b rootcauseCatalogBody) Read(p []byte) (int, error) {
	n, e := b.ReadCloser.Read(p)
	b.metric.Bytes += n
	b.metric.TotalMS = time.Since(b.began).Milliseconds()
	if e != nil {
		b.metric.Cause = rootcauseCatalogError(b.ctx, e)
	}
	return n, e
}

func TestRootCausePublicSearchNetwork(t *testing.T) {
	if os.Getenv("YTMGO_ROOTCAUSE_PUBLIC_NETWORK") != "1" {
		t.Skip("explicit public network diagnosis not selected")
	}
	type result struct {
		Case    int                     `json:"case"`
		Success bool                    `json:"success"`
		Items   int                     `json:"items"`
		TotalMS int64                   `json:"total_ms"`
		Request *rootcauseCatalogMetric `json:"request"`
	}
	var rows []result
	for index, q := range []string{"P!nk Try Official Video", "Try Pink", "P!nk Try Official Video", "Get Lucky"} {
		metric := &rootcauseCatalogMetric{}
		c := NewWithOptions(Options{HTTPClient: &http.Client{Transport: roundTripFunc(func(req *http.Request) (*http.Response, error) {
			began := time.Now()
			res, e := defaultTransport().RoundTrip(req)
			metric.HeaderMS = time.Since(began).Milliseconds()
			metric.Cause = rootcauseCatalogError(req.Context(), e)
			if res != nil {
				metric.Status = res.StatusCode
				res.Body = rootcauseCatalogBody{res.Body, req.Context(), began, metric}
			}
			return res, e
		})}})
		ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
		began := time.Now()
		page, err := c.Search(ctx, q)
		cancel()
		rows = append(rows, result{index + 1, err == nil, len(page.Items), time.Since(began).Milliseconds(), metric})
		if path := os.Getenv("YTMGO_LIVE_EVIDENCE"); path != "" {
			b, _ := json.MarshalIndent(rows, "", "  ")
			_ = os.WriteFile(path, append(b, '\n'), 0600)
		}
	}
}
